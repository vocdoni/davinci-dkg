\begin{tikzpicture}[
  font=\small,

  lifeline/.style={draw, rounded corners=2pt, minimum width=2.8cm, minimum height=8mm, align=center, fill=white},

  phase/.style={font=\bfseries},
  phasebox/.style={draw=gray!50, fill=gray!5, rounded corners=1pt, inner sep=3pt, anchor=west},
  note/.style={font=\footnotesize, align=left}
]

% === Spacing constants ===
\def\colA{0}        % Organizer x
\def\colB{4.5}      % Participants x

\node[lifeline] (nodes) at (\colB, \headerY) {Participants};
\node[lifeline] (chain) at (\colC, \headerY) {Blockchain};

% Vertical dashed lifelines
\draw[draw, dashed, gray] (\colA, \startY + 0.3) -- (\colA, \lifelineEnd);
\draw[draw, dashed, gray] (\colB, \startY + 0.3) -- (\colB, \lifelineEnd);
\draw[draw, dashed, gray] (\colC, \startY + 0.3) -- (\colC, \lifelineEnd);

% === Phase 1 — Initiation =====================================================

\def\pX{-1.5}

\node[phasebox, phase] at (-1.2, \pX) {Phase 1 — Initiation};

\draw[tx] (\colA, \pX - 0.8) -- node[above, font=\footnotesize]{publish $(t,n)$ and policy parameters} (\colC, \pX - 0.8);

\draw[read] (\colA, \pX - 1.4) -- node[above, font=\footnotesize]{rid} (\colB, \pX - 1.4);

\draw[read] (\colB, \pX - 1.4) -- node[above, font=\footnotesize]{rid and process info} (\colC, \pX - 1.4);

\draw[tx] (\colB, \pX - 0.6) -- node[above, font=\footnotesize]{readiness signals} (\colC, \pX - 0.6);

% Separator
%\draw[gray!40, thin] (-1.5, \pX - 0.9) -- (11, \pX - 0.9);

% === Phase 3 — Determine Participants =========================================

\pgfmathparse{\pX - 1.7} 
\edef\pX{\pgfmathresult}

\node[phasebox, phase] at (-1.2, \pX) {Phase 3 — Determine Participants};

% Organizer triggers computation
\draw[tx] (\colA, \pX - 0.9) -- node[above, font=\footnotesize]{\texttt{computeParticipants()}} (\colC, \pX - 0.9);

% Blockchain self-loop: compute I
\draw[tx] (\colC, \pX - 1.1) 
    to[out=0, in=0, looseness=2.0] 
    node[right, pos=0.5, font=\footnotesize, align=left]{compute\\participanting\\nodes} 
    (\colC, \pX - 1.7);

% Participants read the result
\draw[read] (\colB, \pX - 1.9) -- node[above, font=\footnotesize]{read part. nodes} (\colC, \pX - 1.9);

% === Phase 4 — Main DKG =======================================================

\pgfmathparse{\pX - 2.7} 
\edef\pX{\pgfmathresult}

\node[phasebox, phase] at (-1.2, \pX) {Phase 4 — Main DKG};

% Square version
\draw[tx] (\colB, \pX - 0.5) 
    -- ++(-1.2, 0) 
    -- ++(0, -0.8) 
    -- ++(1.2, 0);
\node[left, font=\footnotesize, align=right] at (\colB - 1.2, \pX - 0.9) 
    {compute\\$C_i$, $E_i$, $\pi_i$};

% Single combined submission
\draw[tx] (\colB, \pX - 1.6) -- node[above, font=\footnotesize]{submit $C_i$, $E_i$, $\pi_i$} (\colC, \pX - 1.6);

% Blockchain verifies
\draw[tx] (\colC, \pX - 1.8) 
    to[out=0, in=0, looseness=2.0] 
    node[right, pos=0.5, font=\footnotesize, align=left]{verify inputs} 
    (\colC, \pX - 2.4);

\pgfmathparse{\pX - 3.2}

% Participants read data from the s.c.
